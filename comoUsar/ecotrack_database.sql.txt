create database ecotrack;
use ecotrack;

CREATE TABLE usuarios (
   id_usuario int NOT NULL AUTO_INCREMENT,
   nome_completo varchar(150) NOT NULL,
   email varchar(150) NOT NULL,
   cep varchar(10) NOT NULL,
   endereco varchar(255) DEFAULT NULL,
   telefone varchar(20) NOT NULL,
   cpf varchar(14) NOT NULL,
   senha varchar(255) NOT NULL,
   aceitou_termos tinyint(1) NOT NULL DEFAULT '0',
   deseja_notificacoes tinyint(1) NOT NULL DEFAULT '0',
   data_cadastro timestamp NULL DEFAULT CURRENT_TIMESTAMP,
   firebase_uid varchar(128) DEFAULT NULL,
   `imagem_perfil` varchar(512) DEFAULT NULL,
   `bairro_coleta` VARCHAR(100) DEFAULT NULL,
   `eco_points` INT NOT NULL DEFAULT 0,
   PRIMARY KEY (`id_usuario`),
   UNIQUE KEY `email` (`email`),
   UNIQUE KEY `cpf` (`cpf`),
   UNIQUE KEY `firebase_uid` (`firebase_uid`)
);

CREATE TABLE pontos_coleta (
   id_ponto int NOT NULL AUTO_INCREMENT,
   nome varchar(150) NOT NULL,
   endereco varchar(255) NOT NULL,
   cep varchar(10) DEFAULT NULL,
   latitude decimal(10,8) DEFAULT NULL,
   longitude decimal(11,8) DEFAULT NULL,
   tipo_residuo varchar(255) NOT NULL,
   descricao text,
   horario_funcionamento varchar(100) DEFAULT NULL,
   telefone varchar(20) DEFAULT NULL,
   email varchar(150) DEFAULT NULL,
   PRIMARY KEY (`id_ponto`)
);

CREATE TABLE relatos (
   id_relato int NOT NULL AUTO_INCREMENT,
   id_usuario int NOT NULL,
   titulo varchar(150) NOT NULL,
   bairro varchar(100) NOT NULL,
   endereco varchar(255) NOT NULL,
   categoria varchar(100) NOT NULL,
   descricao text NOT NULL,
   imagem varchar(255) DEFAULT NULL,
   atualizar_status tinyint(1) NOT NULL DEFAULT '0',
   data_criacao timestamp NULL DEFAULT CURRENT_TIMESTAMP,
   status enum('Pendente','Em análise','Resolvido') NOT NULL DEFAULT 'Pendente',
   PRIMARY KEY (id_relato),
   KEY fk_relatos_usuario (id_usuario)
);

CREATE TABLE notificacoes (
   id_notificacao int NOT NULL AUTO_INCREMENT,
   id_usuario int NOT NULL,
   titulo varchar(150) NOT NULL,
   mensagem text NOT NULL,
   lida tinyint(1) NOT NULL DEFAULT '0',
   data_notificacao timestamp NULL DEFAULT CURRENT_TIMESTAMP,
   PRIMARY KEY (`id_notificacao`),
   KEY `id_usuario` (`id_usuario`)
);

CREATE TABLE IF NOT EXISTS notificacoes_push (
    id_notificacao INT NOT NULL,
    estado VARCHAR(20) NOT NULL DEFAULT 'pendente',
    tentativas TINYINT UNSIGNED NOT NULL DEFAULT 0,
    proxima_tentativa DATETIME DEFAULT NULL,
    data_processamento DATETIME DEFAULT NULL,
    ultimo_erro VARCHAR(500) DEFAULT NULL,
    enviada_em DATETIME DEFAULT NULL,
    PRIMARY KEY (id_notificacao),
    KEY idx_notificacoes_push_pendentes (estado, proxima_tentativa)
);

CREATE TABLE IF NOT EXISTS coleta_seletiva_agenda (
    id_agenda INT NOT NULL AUTO_INCREMENT,
    cooperativa VARCHAR(100) NOT NULL,
    dia_semana TINYINT UNSIGNED NOT NULL,
    turno ENUM('Manhã', 'Tarde') NOT NULL,
    areas_coleta TEXT NOT NULL,
    PRIMARY KEY (id_agenda),
    UNIQUE KEY uq_coleta_seletiva_rota (cooperativa, dia_semana, turno),
    KEY idx_coleta_seletiva_dia_turno (dia_semana, turno)
);

CREATE TABLE IF NOT EXISTS coleta_seletiva_lembretes (
    id_usuario INT NOT NULL,
    id_agenda INT NOT NULL,
    data_coleta DATE NOT NULL,
    PRIMARY KEY (id_usuario, id_agenda, data_coleta),
    CONSTRAINT fk_lembrete_coleta_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios (id_usuario)
        ON DELETE CASCADE,
    CONSTRAINT fk_lembrete_coleta_agenda
        FOREIGN KEY (id_agenda) REFERENCES coleta_seletiva_agenda (id_agenda)
        ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS reciclagens (
    id_reciclagem INT NOT NULL AUTO_INCREMENT,
    id_usuario INT NOT NULL,
    material VARCHAR(50) NOT NULL,
    quantidade_kg DECIMAL(8,2) NOT NULL,
    data_reciclagem DATE NOT NULL,
    local_reciclagem VARCHAR(255) NOT NULL,
    comprovante VARCHAR(255) NOT NULL,
    status ENUM('Pendente', 'Aprovada', 'Rejeitada') NOT NULL DEFAULT 'Pendente',
    data_registro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id_reciclagem),
    KEY idx_reciclagens_usuario_status (id_usuario, status),
    CONSTRAINT fk_reciclagens_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios (id_usuario)
        ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS desafios (
    id_desafio INT NOT NULL AUTO_INCREMENT,
    codigo VARCHAR(50) NOT NULL,

    meta INT UNSIGNED NOT NULL,
    recompensa INT UNSIGNED NOT NULL,
    PRIMARY KEY (id_desafio),
    UNIQUE KEY uq_desafios_codigo (codigo)
);

CREATE TABLE IF NOT EXISTS usuarios_desafios (
    id_usuario INT NOT NULL,
    id_desafio INT NOT NULL,
    progresso INT UNSIGNED NOT NULL DEFAULT 0,
    data_conclusao DATETIME DEFAULT NULL,
    PRIMARY KEY (id_usuario, id_desafio),
    CONSTRAINT fk_usuarios_desafios_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios (id_usuario)
        ON DELETE CASCADE,
    CONSTRAINT fk_usuarios_desafios_desafio
        FOREIGN KEY (id_desafio) REFERENCES desafios (id_desafio)
        ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS reciclagens_contabilizadas (
    id_reciclagem INT NOT NULL,
    id_usuario INT NOT NULL,
    data_aprovacao DATETIME NOT NULL,
    PRIMARY KEY (id_reciclagem),
    CONSTRAINT fk_reciclagens_contabilizadas_reciclagem
        FOREIGN KEY (id_reciclagem) REFERENCES reciclagens (id_reciclagem)
        ON DELETE CASCADE,
    CONSTRAINT fk_reciclagens_contabilizadas_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios (id_usuario)
        ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS notificacoes_push (
    id_notificacao INT NOT NULL,
    estado VARCHAR(20) NOT NULL DEFAULT 'pendente',
    tentativas TINYINT UNSIGNED NOT NULL DEFAULT 0,
    proxima_tentativa DATETIME DEFAULT NULL,
    data_processamento DATETIME DEFAULT NULL,
    ultimo_erro VARCHAR(500) DEFAULT NULL,
    enviada_em DATETIME DEFAULT NULL,
    PRIMARY KEY (id_notificacao),
    KEY idx_notificacoes_push_pendentes (estado, proxima_tentativa)
);

CREATE TABLE IF NOT EXISTS expo_push_tokens (
    id_token INT NOT NULL AUTO_INCREMENT,
    id_usuario INT NOT NULL,
    expo_push_token VARCHAR(255) NOT NULL,
    plataforma ENUM('ios', 'android') NOT NULL,
    ativo TINYINT(1) NOT NULL DEFAULT 1,
    data_registro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id_token),
    UNIQUE KEY uq_expo_push_token (expo_push_token),
    KEY idx_expo_push_usuario_ativo (id_usuario, ativo),
    CONSTRAINT fk_expo_push_tokens_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios (id_usuario)
        ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS usuarios_pontos_coleta (
    id_usuario INT NOT NULL,
    id_ponto INT NOT NULL,
    data_acompanhamento TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id_usuario, id_ponto),
    CONSTRAINT fk_usuario_ponto_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios (id_usuario)
        ON DELETE CASCADE,
    CONSTRAINT fk_usuario_ponto_ponto
        FOREIGN KEY (id_ponto) REFERENCES pontos_coleta (id_ponto)
        ON DELETE CASCADE
);
